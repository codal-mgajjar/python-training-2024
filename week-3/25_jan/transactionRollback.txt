

DROP TABLE IF EXISTS users;
CREATE TABLE users(
 id INT GENERATED BY DEFAULT AS IDENTITY,
 name VARCHAR(50),
 gems DEC (10),
 PRIMARY KEY (id));

BEGIN;
UPDATE users
SET gems = gems - 100
WHERE id=2;
UPDATE users
SET gems = gems+100
WHERE id=1;
COMMIT;
SELECT * FROM users;

# rollback impliment

BEGIN;
UPDATE users
SET gems = gems - 100
WHERE id=2;
UPDATE users
SET gems = gems+100
WHERE id=1;
SELECT * FROM users;
ROLLBACK;
SELECT * FROM users;

#Savepoint impliment
BEGIN;
UPDATE users
SET gems = gems - 100
WHERE id=2;
UPDATE users
SET gems = gems+100
WHERE id=1;
SAVEPOINT bijo;
SELECT * FROM users;

ROLLBACK TO SAVEPOINT pelo;
SELECT * from users;
